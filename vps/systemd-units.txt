### trade-bot.service
[Unit]
Description=Trade Journal Telegram bot
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=ubuntu
WorkingDirectory=/home/ubuntu/trade-bot
Environment=BOT_MODE=polling
Environment=PORT=3000
Environment=WORKER_URL=http://127.0.0.1:3100
ExecStart=/usr/bin/node bot.js
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target

### tv-worker.service
[Unit]
Description=TradingView snapshot worker
After=tv-browser.service
Wants=tv-browser.service

[Service]
Environment=WORKER_PORT=3100
Environment=DXY_SYMBOL=CAPITALCOM:DXY
User=ubuntu
WorkingDirectory=/home/ubuntu/tv-worker
ExecStart=/usr/bin/node worker.js
Restart=always
RestartSec=5
# Страховка от утечки: 25.09 воркер раздулся до 1.45 ГБ на брошенных CDP-подключениях
# и утянул за собой браузер. Теперь при раздутии его убьёт systemd и сразу поднимет
# заново — это потеря одной съёмки вместо потери съёмщика на двое суток.
MemoryMax=1200M

[Install]
WantedBy=multi-user.target

### tv-browser.service
[Unit]
Description=Persistent Chromium with TradingView session
After=xvfb.service fluxbox.service
Requires=xvfb.service

[Service]
Environment=DISPLAY=:99
Environment=HOME=/home/ubuntu
User=ubuntu
WorkingDirectory=/home/ubuntu/tv-worker
# Chromium от snap живёт в пользовательском scope systemd, а не в cgroup сервиса.
# Если он умер грязно, в профиле остаётся замок, и каждый следующий запуск молча
# уходит в «Opening in existing browser session» — 23.09 так получился цикл из 96
# перезапусков и сделка без скринов. Перед стартом подчищаем хвосты.
ExecStartPre=-/usr/bin/pkill -u ubuntu -f "chromium.*tv-profile"
ExecStartPre=-/bin/sh -c "rm -f /home/ubuntu/tv-profile/Singleton*"
ExecStart=/usr/bin/node browser.js
Restart=always
RestartSec=10

[Install]
WantedBy=multi-user.target


### tv-browser-restart.service + .timer
# Chromium, работающий неделями, деградирует незаметно. Раз в сутки поднимаем его
# заново — в 02:30 UTC (05:30 по Минску), до начала лондонской сессии и после окна
# обновлений snap (01:00-02:00 UTC).

# /etc/systemd/system/tv-browser-restart.service
[Unit]
Description=Nightly restart of the TradingView browser

[Service]
Type=oneshot
ExecStart=/usr/bin/systemctl restart tv-browser.service

# /etc/systemd/system/tv-browser-restart.timer
[Unit]
Description=Nightly restart of the TradingView browser

[Timer]
OnCalendar=*-*-* 02:30:00
Persistent=true

[Install]
WantedBy=timers.target
