### trade-bot.service
[Unit]
Description=Trade Journal Telegram bot
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=ubuntu
WorkingDirectory=/home/ubuntu/trade-bot
Environment=BOT_MODE=polling
Environment=PORT=3000
Environment=WORKER_URL=http://127.0.0.1:3100
ExecStart=/usr/bin/node bot.js
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target

### x11vnc.service
[Unit]
Description=VNC for :99 (localhost only)
After=xvfb.service
Requires=xvfb.service
[Service]
ExecStart=/usr/bin/x11vnc -display :99 -forever -shared -rfbport 5900 -localhost -nopw -quiet
Restart=always
[Install]
WantedBy=multi-user.target

### novnc.service
