### trade-bot.service
[Unit]
Description=Trade Journal Telegram bot
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=ubuntu
WorkingDirectory=/home/ubuntu/trade-bot
Environment=BOT_MODE=polling
Environment=PORT=3000
Environment=WORKER_URL=http://127.0.0.1:3100
ExecStart=/usr/bin/node bot.js
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
